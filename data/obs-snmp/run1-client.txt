### $ # T4 (client): T3 の 1 回目のあとに実行。約 20 秒、バックグラウンドで動く   (client)
exit=0

client
exit=0
### $ ping -c 100 -i 0.2 -s 1400 192.168.32.20 > ~/ping-snmp.txt 2>&1 < /dev/null &   (client)

exit=0
